157 private static LocalTime parseTime(String raw) {
158 if (raw == null || raw.isBlank()) return null;
159 String r = raw.trim();
160 r = r.replace('.', ':').replace('-', ':');
161 for (DateTimeFormatter f : TIME_FORMATS) {
162 try {
163 return LocalTime.parse(r, f);
164 } catch (DateTimeParseException ignored) {}
165 }
166 if (r.matches("^\\d{1,2}$")) {
167 try { return LocalTime.of(Integer.parseInt(r), 0); } catch (Exception ignored) {}
168 }
169 if (r.matches("^\\d{3,4}$")) {
170 try {
171 int val = Integer.parseInt(r);
172 int hour = val / 100; int min = val % 100;
173 if (hour >=0 && hour <= 23 && min >=0 && min <=59) return LocalTime.of(hour, min);
174 } catch (Exception ignored) {}
175 }
176 return null;
177 }
