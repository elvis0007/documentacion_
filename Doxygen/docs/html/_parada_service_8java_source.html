109 private static LocalTime parseTime(String raw) {
110 if (raw == null || raw.isBlank()) return null;
111 String r = raw.trim();
112 r = r.replace('.', ':').replace('-', ':');
113 for (DateTimeFormatter f : TIME_FORMATS) {
114 try {
115 return LocalTime.parse(r, f);
116 } catch (DateTimeParseException ignored) {}
117 }
118 if (r.matches("^\\d{1,2}$")) {
119 try { return LocalTime.of(Integer.parseInt(r), 0); } catch (Exception ignored) {}
120 }
121 if (r.matches("^\\d{3,4}$")) {
122 try {
123 int val = Integer.parseInt(r);
124 int hour = val / 100; int min = val % 100;
125 if (hour >=0 && hour <= 23 && min >=0 && min <=59) return LocalTime.of(hour, min);
126 } catch (Exception ignored) {}
127 }
128 return null;
